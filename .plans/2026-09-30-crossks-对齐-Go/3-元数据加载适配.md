    use astersql_infoschema::builder::{ActionType, Builder, MetadataReader, SchemaDiff};
    use astersql_infoschema::infoschema::{CiString, ColumnInfo, DBInfo, TableInfo};
    use std::sync::Arc;
    struct Reader(TableInfo);
    impl MetadataReader for Reader {
        fn database(&self, _: i64) -> Result<Option<DBInfo>, String> { Ok(None) }
        fn table(&self, _: i64, _: i64) -> Result<Option<TableInfo>, String> { Ok(Some(self.0.clone())) }
    }
    fn main() {
        let column = |id| ColumnInfo { id, name: CiString::new(format!("c{id}")), ..Default::default() };
        let old = TableInfo { id: 100, db_id: 1, name: CiString::new("system_table"), columns: vec![column(1)], ..Default::default() };
        let mut db = DBInfo { id: 1, name: CiString::new("mysql"), tables: vec![Arc::new(old.clone())], ..Default::default() };
        let mut builder = Builder::new(astersql_infoschema::infoschema_v2::NewData(), false).WithCrossKS(true);
        builder.InitWithDBInfos(std::slice::from_mut(&mut db), vec![], vec![], 1);
        let mut new = old;
        new.columns.push(column(2));
        let ids = builder.ApplyDiff(&Reader(new), &SchemaDiff { version: 2, action_type: ActionType::AddColumn, schema_id: 1, table_id: 100, ..Default::default() }).unwrap();
        let schema = builder.Build(2);
        let table = schema.TableByName(&CiString::new("mysql"), &CiString::new("system_table")).unwrap();
        println!("schema_version={} column_count={} affected={ids:?}", schema.SchemaMetaVersion(), table.Meta().columns.len());
        assert_eq!(table.Meta().columns.len(), 2, "AddColumn must refresh table metadata");
    }
